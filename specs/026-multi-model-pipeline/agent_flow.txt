@startuml Agent Flow
skinparam conditionStyle InsideDiamond
skinparam SequenceGroupBodyBackgroundColor #cccccc

header Generative Creative Lab
title Multi-Agent Adaptation Flow
footer GitHub Ticket 26

actor "User" as HU
database "Django" as DB
queue "Adaptation Queue" as AQ
entity "Supervisor Agent" as SA
entity "Concept Agent" as CDA #337cf2
entity "Culture Agent" as CUA #e5cc38
entity "Writer Agent" as WRA #cc3fef

== Adaptation Request ==

activate HU
HU -> DB: 1.0.0 Requests New Adaptation
activate DB
DB -> AQ: 1.1.0 Adds Task to Queue
activate AQ
AQ -> SA: 1.2.0 Launch Task [Payload]
activate SA
AQ -[#purple]-> DB: <color:purple>1.3.0 Status="Pending"</color>

== Task Start ==

group#337cf2 #cfdfff <color:white>Concept Inferrence</color>
    SA -> CDA: 2.0.0 Invoke SubAgent
    activate CDA
    SA -[#purple]> DB: <color:purple>2.1.0 Status="Concept Analysis"</color>
    CDA --> SA: 2.2.0 Return Concept Brief
    deactivate CDA
end

group#e5cc38 #e5dfae Cultural Inferrence
    SA -> CUA: 3.0.0 Invoke SubAgent
    activate CUA
    SA -[#purple]-> DB: <color:purple>3.1.0 Status="Cultural Analysis"</color>
    CUA --> SA: 3.2.0 Return Cultural Brief
    deactivate CUA
end

group#cc3fef #f7cfff <color:white>Writing Inferrence</color>
    SA -> WRA: 4.0.0 Invoke SubAgent
    activate WRA
    SA -[#purple]-> DB: <color:purple>4.1.0 Status="Writing"</color>
    WRA --> SA: 4.2.0 Return Adapted Script
    deactivate WRA
end

group#333 #333 <color:white>Evaluation Process</color>
    loop#e5cc38 #e5dfae Cultural Alignment Review: Max 3 Revisions
        SA -> CUA: 5.1.0 Evaluate Script for Concept Alignment
        activate CUA
        CUA --> SA: 5.1.1 Return Feedback
        deactivate CUA
        alt#66ff66 #99ff99 Concept Aligned
            SA -> DB: Status="Concept Aligned"
        else #fcc Cultural Revision Needed
            SA -> WRA: 5.1.2 Request Revision
            activate WRA
            SA -[#purple]> DB: <color:purple>5.1.3 Status="Revising for Cultural Alignment"</color>
            WRA --> SA: 5.1.4 Return Revised Script
            deactivate WRA
        end
    end
    loop#337cf2 #cfdfff Concept Alignment Review: Max 3 Revisions
        SA -> CDA: 5.2.0 Evaluate Script for Concept Alignment
        activate CDA
        CDA --> SA: 5.2.1 Return Feedback
        deactivate CDA
        alt#66ff66 #99ff99 Concept Aligned
            SA -> DB: Status="Concept Aligned"
        else #fcc Concept Revision Needed
            SA -> WRA: 5.2.2 Request Revision
            activate WRA
            SA -[#purple]> DB: <color:purple>5.2.3 Status="Revising for Concept"</color>
            WRA --> SA: 5.2.4 Return Revised Script
            deactivate WRA
        end
    end

    break#f99 #fcc Revision Limit Exceeded
        SA -[#red]-> DB: <color:red>5.3.0 Status="Failed"</color>
    end
end

SA -[#green]-> AQ: <color:green>6.0.0 End Task with Status
deactivate SA

== Adaptation Task End ==

HU --> DB: 6.1.0 Reviews Adaptation
deactivate HU
